procedure proc_nafhead(a_entity_code varchar2,
a_tcode varchar2,
a_tnature varchar2,
a_acc_year varchar2) is
L_DECIMAL NUMBER := lhs4_client.GET_AMT_DECIMAL(a_entity_code);
L_SUM_AFIELD1 NUMBER := 0;
L_SLNO NUMBER := 0;
L_VRNO VARCHAR2(11);
L_TOTAL_AMT NUMBER := 0;
L_AMT NUMBER := 0;
l_yrbegdate date;
L_YRENDDATE DATE;
l_n1 NUMBER := 0;
l_n2 NUMBER := 0;
l_nafield NUMBER := 0;
CURSOR C_VR IS
SELECT ENTITY_CODE, TCODE, VRNO, NAFHEAD, NAFIELD
FROM NAFHEAD_TRAN
WHERE ENTITY_CODE = a_ENTITY_CODE
AND TCODE = a_TCODE
AND SUBSTR(VRNO, 3, 2) = SUBSTR(a_ACC_YEAR, 1, 2)
AND TNATURE = a_TNATURE
AND NVL(ALLOC_TYPE, &#39;V&#39;) = &#39;V&#39;;
CURSOR C_P IS
SELECT ENTITY_CODE, TCODE, VRNO, NAFHEAD, NAFIELD, FROM_DATE,
TO_DATE
FROM NAFHEAD_TRAN
WHERE ENTITY_CODE = a_ENTITY_CODE
AND TCODE = a_TCODE
AND TNATURE = a_TNATURE
AND FROM_DATE BETWEEN l_YRBEGDATE AND l_YRENDDATE
AND TO_DATE BETWEEN l_YRBEGDATE AND l_YRENDDATE
AND NVL(ALLOC_TYPE, &#39;V&#39;) &lt;&gt; &#39;V&#39;
ORDER BY NAFHEAD, FROM_DATE;
BEGIN
select yrbegdate, YRENDDATE
into l_yrbegdate, L_YRENDDATE
from acc_year_mast
where entity_code = a_entity_code
and acc_year = a_acc_year;
FOR I IN C_VR LOOP
SELECT SUM(AFIELD1)
INTO L_SUM_AFIELD1
FROM itemTRAN_BODY
WHERE ENTITY_CODE = I.ENTITY_CODE
AND TCODE = I.TCODE
AND VRNO = I.VRNO;
L_TOTAL_AMT := 0;

DECLARE
CURSOR C2 IS
SELECT VRNO, SLNO, AFIELD1, vrdate
FROM itemTRAN_BODY
WHERE ENTITY_CODE = a_ENTITY_CODE
AND TCODE = a_TCODE
AND VRNO = I.VRNO;
BEGIN
FOR J IN C2 LOOP
L_AMT := ROUND((I.NAFIELD + J.AFIELD1) / L_SUM_AFIELD1,
L_DECIMAL);
L_TOTAL_AMT := NVL(L_TOTAL_AMT, 0) + NVL(L_AMT, 0);
L_SLNO := J.SLNO;
END LOOP;
IF L_TOTAL &lt;&gt; I.NAFIELD THEN
NULL;
END IF;
END;
END LOOP;
